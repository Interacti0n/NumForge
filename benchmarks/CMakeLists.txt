# Opt-in measurement tools. Installed math/application targets remain unchanged.
add_library(numforge_benchmark_clock OBJECT common/benchmark_clock.c)
target_include_directories(numforge_benchmark_clock PRIVATE "${PROJECT_SOURCE_DIR}/include")
numforge_enable_diagnostics(numforge_benchmark_clock)

function(numforge_benchmark target layer source)
    add_executable(${target} ${source} ${ARGN})
    target_link_libraries(${target} PRIVATE ${layer} numforge_benchmark_clock)
    # Preserve the common binary-directory argument used by the recorders.
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
    numforge_enable_diagnostics(${target})
endfunction()

if(BUILD_TESTING)
    function(numforge_benchmark_check name timeout)
        add_test(NAME ${name} COMMAND ${ARGN})
        set_tests_properties(${name} PROPERTIES TIMEOUT ${timeout})
    endfunction()
    foreach(target IN ITEMS numerical_benchmark numerical_scan_benchmark
            matrix_benchmark statistics_benchmark polynomial_benchmark)
        numforge_benchmark_check(${target}_smoke 120 ${target})
    endforeach()
    foreach(kernel IN ITEMS grid refinement curve phase)
        numforge_benchmark_check(numerical_sampling_${kernel}_smoke 120 numerical_sampling_benchmark ${kernel})
    endforeach()
    numforge_benchmark_check(allocation_stats_check 30 allocation_stats_check)
    numforge_benchmark_check(cache_memory_benchmark_smoke 120 cache_memory_benchmark --quick)
    numforge_benchmark_check(decimal_format_benchmark_smoke 120 decimal_format_benchmark --quick)
    numforge_benchmark_check(graph_cache_benchmark_smoke 120 graph_verify_benchmark cache 1)
    numforge_benchmark_check(graph_refinement_benchmark_smoke 120 graph_verify_benchmark refinement 1)
    numforge_benchmark_check(numeric_root_scan_benchmark_smoke 120 numeric_root_scan_benchmark --quick)
    foreach(method IN ITEMS bisection brent hybrid)
        set(options)
        if(method STREQUAL "brent")
            set(options --compare)
        elseif(method STREQUAL "hybrid")
            set(options --hybrid)
        endif()
        numforge_benchmark_check(numeric_root_${method}_smoke 120 numeric_root_benchmark ${options} --quick)
        numforge_benchmark_check(numeric_root_${method}_domain_regression 120 numeric_root_benchmark ${options} --adversarial)
    endforeach()

    # These are correctness/failure tests, not disposable performance recordings.
    target_compile_definitions(logarithm_agm_benchmark PRIVATE NUMFORGE_ENABLE_ALLOC_FAILURE_TESTING)
    target_compile_definitions(bigint_arithmetic_benchmark PRIVATE NUMFORGE_ENABLE_ALLOC_FAILURE_TESTING)
    numforge_benchmark_check(logarithm_agm_fault_checks 300 logarithm_agm_benchmark --fault-checks)
    foreach(shard RANGE 0 3)
        numforge_benchmark_check(logarithm_agm_interval_fault_checks_${shard} 300 logarithm_agm_benchmark --fault-checks-interval ${shard})
    endforeach()
    numforge_benchmark_check(karatsuba_fault_checks 120 bigint_arithmetic_benchmark --karatsuba-check)
    numforge_benchmark_check(multiplication_fault_checks 120 bigint_arithmetic_benchmark --multiply-check)
    numforge_benchmark_check(normalization_fault_checks 120 bigint_arithmetic_benchmark --normalization-check)
    find_program(NUMFORGE_BIGINT_BENCH_NODE NAMES node nodejs)
    if(NUMFORGE_BIGINT_BENCH_NODE)
        foreach(pair IN ITEMS "karatsuba_reference|check_karatsuba.cjs|bigint_arithmetic_benchmark"
                "logarithm_agm_benchmark|check_logarithm_agm_benchmark.cjs|logarithm_agm_benchmark"
                "bigint_arithmetic_benchmark|check_bigint_benchmark.js|bigint_arithmetic_benchmark"
                "decimal_math_benchmark|check_decimal_math_benchmark.js|decimal_math_benchmark")
            string(REPLACE "|" ";" fields "${pair}")
            list(GET fields 0 name)
            list(GET fields 1 checker)
            list(GET fields 2 target)
            numforge_benchmark_check(${name}_smoke 180 "${NUMFORGE_BIGINT_BENCH_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/${checker}" $<TARGET_FILE:${target}>)
        endforeach()
        numforge_benchmark_check(pipeline_benchmark_smoke 30 "${NUMFORGE_BIGINT_BENCH_NODE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_pipeline_benchmark.js"
            $<TARGET_FILE:decimal_math_benchmark> $<TARGET_FILE:calculator_limit_benchmark>)
    elseif(NUMFORGE_REQUIRE_NODE_TESTS)
        message(FATAL_ERROR "Benchmark validation requires Node.js")
    endif()
endif()
